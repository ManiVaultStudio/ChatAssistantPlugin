cmake_minimum_required(VERSION 3.22)
project(AssistantNativeActions LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Network)
find_package(ManiVault COMPONENTS Core CONFIG REQUIRED)
add_executable(assistant-native-actions main.cpp ../../src/ManiVaultControls.cpp)
target_compile_features(assistant-native-actions PRIVATE cxx_std_20)
target_compile_definitions(assistant-native-actions PRIVATE ASSISTANT_MANIVAULT_ACTIONS=1)
target_include_directories(assistant-native-actions PRIVATE ../../src "${ManiVault_INCLUDE_DIR}")
target_link_libraries(assistant-native-actions PRIVATE Qt6::Widgets Qt6::Sql Qt6::Network ManiVault::Core)
